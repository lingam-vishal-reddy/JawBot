[context]
Skill: Chromium for Android (plain-text playbook)

Goal: fetch depot_tools, check out Chromium source configured for Android, and
build the target APK.

Assumptions / environment:
- Linux host with git, python3, and enough disk (Chromium needs 100GB+).
- depot_tools lives at ${DEPOT_TOOLS:-$HOME/depot_tools} and must be on PATH.
- Chromium source lives at ${CHROMIUM_SRC:-$HOME/chromium/src}.
- Override with CHROMIUM_ROOT, CHROMIUM_SRC, DEPOT_TOOLS env vars.

Use the params for specifics: check out {branch}, generate GN files into
{output_dir} with target_os={target_os} and is_debug={is_debug}, and build
{apk_target}. The user's message may override any of these.

Status: setup and build run step by step; ask "status" for progress.

[task setup]
name: Chromium Android setup
description: Install depot_tools and fetch/sync a Chromium Android checkout on the chosen branch.
triggers: setup, set up, install, fetch, prepare, sync

step: Clone depot_tools
template: if [ ! -d "${DEPOT_TOOLS:-$HOME/depot_tools}" ]; then git clone https://chromium.googlesource.com/chromium/tools/depot_tools.git "${DEPOT_TOOLS:-$HOME/depot_tools}"; else echo "depot_tools already present"; fi
timeoutMs: 600000

step: Fetch Chromium (Android)
template: export PATH="${DEPOT_TOOLS:-$HOME/depot_tools}:$PATH"; mkdir -p "${CHROMIUM_ROOT:-$HOME/chromium}"; cd "${CHROMIUM_ROOT:-$HOME/chromium}"; if [ ! -d "${CHROMIUM_SRC:-$HOME/chromium/src}/.git" ]; then fetch --nohooks android; else echo "chromium checkout already present"; fi
timeoutMs: 3600000

step: Install build deps
template: export PATH="${DEPOT_TOOLS:-$HOME/depot_tools}:$PATH"; cd "${CHROMIUM_SRC:-$HOME/chromium/src}"; ./build/install-build-deps.sh --android || true
timeoutMs: 1800000

step: Checkout branch and sync
template: export PATH="${DEPOT_TOOLS:-$HOME/depot_tools}:$PATH"; cd "${CHROMIUM_SRC:-$HOME/chromium/src}"; git checkout main 2>/dev/null || true; gclient sync
timeoutMs: 3600000

[task build]
name: Chromium Android build
description: Configure GN args for Android and build the APK target.
triggers: build, compile, make, apk

step: gn gen (Android)
template: export PATH="${DEPOT_TOOLS:-$HOME/depot_tools}:$PATH"; cd "${CHROMIUM_SRC:-$HOME/chromium/src}"; gn gen out/Android --args='target_os="android" is_debug=false'
timeoutMs: 600000

step: autoninja build APK
template: export PATH="${DEPOT_TOOLS:-$HOME/depot_tools}:$PATH"; cd "${CHROMIUM_SRC:-$HOME/chromium/src}"; autoninja -C out/Android chrome_public_apk
timeoutMs: 14400000
